\section{A lower bound for the absolute subgroup growth gradient}\label{section: lower bound subgroup growth gradient}

\begin{thm}\label{thm: subgroup growth gradient}
	Let $G=\langle x_1,\ldots, x_n\mid r_1,\ldots r_m\rangle$ be a finitely presented infinite group with $n>m+1$. Suppose that $G$ is residually finite and that the $m$-tuple $(r_1,\ldots,r_m)$ satisfies a standard minimum condition. Then there exists a chain $(G_k)_{k\ge 0}$ of finite index normal subgroups of $G$ with vanishing intersection such that $\sgg(G,(G_k)_{k\ge 0})=\sgg_{\mathrm{abs}}(G)$ and
	\[
	s(G,\varprojlim G_k)\ge n-m.
	\]
\end{thm}
