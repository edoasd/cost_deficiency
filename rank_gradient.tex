\section{An upper bound for the absolute rank gradient}

\begin{thm}\label{thm: rank gradient}
	Let $G=\langle x_1,\ldots, x_n\mid r_1,\ldots r_m\rangle$ be a finitely presented infinite group with $n>m+1$. Suppose that $G$ is residually finite and that the $m$-tuple $(r_1,\ldots,r_m)$ satisfies a standard minimum condition. Then there exists a chain $(G_k)_{k\ge 0}$ of finite index normal subgroups of $G$ with vanishing intersection such that $\rg(G,(G_k)_{k\ge 0})=\rg_{\mathrm{abs}}(G)$ and
	\[
	\mathrm{cost}(G,\varprojlim G_k)\le n-m.
	\]
\end{thm}
