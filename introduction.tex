\section{Introduction}\label{section: introduction}

\begin{thm}
Let $G$ be a random group in the few relators model with $n\ge 3$ generators and $m\ge 0$ relations. Suppose that $m\le n-2$. Then with asymptotic probability $1$ we have
\[
n-m-1=\mathrm{cost}(G)-1=\rg_{\mathrm{abs}}(G)=\sgg_{\mathrm{abs}}(G)=s(G,\widehat{G})-1.
\]
\end{thm}
